        public async Task<CardTransferQueryResult> CardTransferAsync(CardTransferQuery query)
        {
            var Client = GetClient();
            var token = _businessInfo.ApiToken;
            Client.DefaultRequestHeaders.Authorization =
               new AuthenticationHeaderValue("Bearer", "7245126630-64990800a230406791f049b4e0b55852.XzIwMjUx");

            var dateTime = DateTime.UtcNow.ToString("yyyyMMddHHmmssfff", CultureInfo.InvariantCulture);
            var orgCode = "4321";
            var randomString = _cardTransferSettings.BusinessId + "AB";
            var sumInput = orgCode + randomString + dateTime;

            var total = Encoding.ASCII.GetBytes(sumInput).Sum(b => b);

            var TransactionId = $"{orgCode}-{randomString}-{dateTime}-{total}";
            var MerchantCode = _cardTransferSettings.MerchantCode;
            var TerminalCode = _cardTransferSettings.TerminalCode;

            var requestBody = new
            {
                apiKey = _podroSettings.CardTransferApiKey,
                productEntityId = _podroSettings.CardTransferProductId,
                providerParameters = new
                {
                    body = new
                    {
                        query.SourceCardNumber,
                        query.DestinationCardNumber,
                        query.ClientMobileNumber,
                        query.ClientDeviceId,
                        query.ClientPlatform,
                        query.ClientIP,
                        query.Amount,
                        TerminalCode,
                        query.Pin2,
                        query.Cvv2,
                        query.ExpireMonth,
                        query.ExpireYear,
                        query.DesCardName,
                        query.AuthorizeRefNumber,
                        query.ClientEmail,
                        query.ClientUserAgent,
                        query.TransferFromDesc,
                        query.TransferToDesc,
                        MerchantCode,
                        TransactionId
                    }
                }
            };

            var result = await Client.PostAsJsonAsync<dynamic, PodResponse<PodiumResponse>>("consumers/services/do", requestBody);
            
            if (result == null)
                throw new KidzyException("Response is null");

            if (result.HasError)
                throw new KidzyException(result.Message);

            var serviceResponse = JsonConvert.DeserializeObject<CardTransferView>(result.Result.Result);
            var queryResult = CardTransferQueryResult.Success();
            queryResult.Entity = serviceResponse;
            queryResult.TransactionId = TransactionId;

            return queryResult;
        }
